      if (isTriggered) return;

      cleanup();
      initWaline();

    function handleScroll() {
      if (window.scrollY > 30) {
        hasScrolledDown = true;

          triggerLoad();
        }

      if (e.deltaY > 0) {

      hasScrolledDown = true;

    function setupLazyLoad() {
      const walineEl = document.getElementById('waline');
      if (!walineEl) return;

      walineEl.addEventListener('click', function onClick() {
        walineEl.removeEventListener('click', onClick);
        triggerLoad();

      if (window.location.hash === '#waline-comments' || window.location.hash === '#waline') {
        triggerLoad();
        return;
      }

      // Hash change listener
      window.addEventListener('hashchange', function() {
        if (window.location.hash === '#waline-comments' || window.location.hash === '#waline') {
          triggerLoad();
        }
      });

      // Clicks on table of contents / comment anchor links

        const anchor = e.target && e.target.closest && e.target.closest('a[href*="#waline"]');
        if (anchor) {
          triggerLoad();
        }
      });

      // Listen for actual user scrolling / gestures
      window.addEventListener('scroll', handleScroll, { passive: true });
      window.addEventListener('wheel', handleWheel, { passive: true });
      window.addEventListener('touchmove', handleTouchMove, { passive: true });

      // IntersectionObserver: track proximity to viewport
      if ('IntersectionObserver' in window) {
        observer = new IntersectionObserver(function(entries) {
          for (let i = 0; i < entries.length; i++) {
            if (entries[i].isIntersecting) {
              isNearViewport = true;
              if (hasScrolledDown) {
                triggerLoad();
              }
              break;
            } else {
              isNearViewport = false;
            }
          }
        }, {
          rootMargin: '120px 0px 120px 0px',
          threshold: 0
        });
        observer.observe(walineEl);
      } else {
        // Fallback for older browsers: require scroll
        window.addEventListener('scroll', function() {
          if (window.scrollY > 40) triggerLoad();
        }, { once: true, passive: true });
      }
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', setupLazyLoad, { once: true });
    } else {
      setupLazyLoad();
    }
